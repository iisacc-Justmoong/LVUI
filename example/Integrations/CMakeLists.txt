cmake_minimum_required(VERSION 3.31)
project(LVUIApplicationIntegrations LANGUAGES CXX)

# These are examples of application-owned composition; neither SDK package
# exports or acquires the other as a dependency.
option(APPLICATION_COLOR_ADAPTER "Compose iiColorManager in the application" ON)
option(APPLICATION_IMAGE_ADAPTER "Compose image I/O services in the application" OFF)
if(APPLICATION_COLOR_ADAPTER)
    find_package(LVUI CONFIG REQUIRED COMPONENTS Color)
    find_package(iiColorManager CONFIG REQUIRED COMPONENTS iiColorManager)
    add_library(ApplicationColorAdapter STATIC App/ColorSelectionSurface.cpp)
    target_compile_features(ApplicationColorAdapter PUBLIC cxx_std_23)
    target_link_libraries(ApplicationColorAdapter PUBLIC LVUI::Color PRIVATE iiColorManager::iiColorManager)
endif()
if(APPLICATION_IMAGE_ADAPTER)
    find_package(LVUI CONFIG REQUIRED COMPONENTS ImageIO)
    find_package(iiSDKCore 0.20 CONFIG REQUIRED COMPONENTS Runtime)
    find_package(iiGUI 0.5.0 CONFIG REQUIRED)
    find_package(iiFileProviderNative 0.9 CONFIG REQUIRED)
    find_package(iiNetwork 0.5 CONFIG REQUIRED)
    find_package(iiVisualCodec 0.2 CONFIG REQUIRED COMPONENTS Image)
    find_package(CURL 7.62 REQUIRED)
    find_package(ZLIB REQUIRED)
    add_library(ApplicationImageIOAdapter STATIC App/NativeImageLoader.cpp App/NativeImageSource.cpp)
    target_compile_features(ApplicationImageIOAdapter PUBLIC cxx_std_23)
    target_include_directories(ApplicationImageIOAdapter PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/App")
    target_link_libraries(ApplicationImageIOAdapter PUBLIC LVUI::ImageIO iiGUI::iiGUI iiSDKCore::Runtime
        PRIVATE iiFileProvider::Native iiNetwork::iiNetwork iiVisualCodec::Image CURL::libcurl ZLIB::ZLIB)
endif()
