cmake_minimum_required(VERSION 3.31)
project(LVUIStyleTokenConsumer LANGUAGES CXX)
include(CTest)
find_package(LVUI 0.1 CONFIG REQUIRED)
find_package(LVRS CONFIG REQUIRED COMPONENTS Native)
if(NOT LVRS_NATIVE)
    message(FATAL_ERROR "The style token example requires the Qt-free native LVRS kit")
endif()
lvui_generate(NAME TokenWindow SOURCE App/TokenWindow.lvui FRAMEWORK LVRS
    STYLE_MODES "Theme Colors=Dark" "Theme Typography=Value" OUTPUT_VARIABLE _window)
lvui_generate(NAME InlineTokens SOURCE App/InlineTokens.lvui FRAMEWORK LVRS OUTPUT_VARIABLE _inline)
add_executable(LVUIStyleTokenConsumer App/main.cpp "${_window}" "${_inline}")
target_compile_features(LVUIStyleTokenConsumer PRIVATE cxx_std_23)
target_include_directories(LVUIStyleTokenConsumer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(LVUIStyleTokenConsumer PRIVATE LVRS::LVRS)
add_executable(LVUITokenCatalogConsumer App/catalog.cpp)
target_link_libraries(LVUITokenCatalogConsumer PRIVATE LVUI::LVUI)
if(BUILD_TESTING)
    add_test(NAME LVUI.StyleTokens.InstalledCatalog COMMAND LVUITokenCatalogConsumer)
    add_test(NAME LVUI.StyleTokens.CompiledAppearance COMMAND LVUIStyleTokenConsumer)
    option(LVUI_STYLE_TOKEN_NATIVE_TESTS "Open a short-lived native window and verify GPU pixels" OFF)
    if(LVUI_STYLE_TOKEN_NATIVE_TESTS)
        add_test(NAME LVUI.StyleTokens.Native COMMAND LVUIStyleTokenConsumer native)
        add_test(NAME LVUI.StyleTokens.Vulkan COMMAND LVUIStyleTokenConsumer vulkan)
        set_tests_properties(LVUI.StyleTokens.Native LVUI.StyleTokens.Vulkan PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
    endif()
endif()
