cmake_minimum_required(VERSION 3.31)
project(LVUIQueryConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(LVUI 0.1 CONFIG REQUIRED COMPONENTS Native)
find_package(iiSDKCore 0.20 CONFIG REQUIRED COMPONENTS Runtime)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE LVUI::Native iiSDKCore::Runtime)
enable_testing()
add_test(NAME LVUI.QueryConsumer COMMAND consumer)
