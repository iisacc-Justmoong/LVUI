cmake_minimum_required(VERSION 3.31)
project(LVUIInstalledApplication LANGUAGES CXX)
set(_lvui_components Native)
if(LOCALIZED_APPLICATION)
    list(APPEND _lvui_components Localization)
endif()
find_package(LVUI CONFIG REQUIRED COMPONENTS ${_lvui_components})
find_package(iiGUI 0.5.0 CONFIG REQUIRED)

if(BAD_APPLICATION_TARGET)
    lvui_add_application(bad-target SOURCE NativeApplication.lvui)
elseif(BAD_CONFIGURE_SOURCE)
    lvui_add_application(InvalidConfigureSource SOURCE NativeApplication.lvui
        CONFIGURE_SOURCE does-not-exist.cpp)
else()
    if(LOCALIZED_APPLICATION)
        # The application, rather than the LVUI package, chooses these adapters.
        find_package(iiJson CONFIG REQUIRED)
        find_package(iiFileProviderNative CONFIG REQUIRED)
        lvui_add_application(InstalledLocalized NAME LocalizedApplication
            SOURCE LocalizedApplication.lvui CONFIGURE_SOURCE LocalizedApplication.cpp
            LIBRARIES LVUI::Localization iiJson::iiJson iiFileProvider::Native)
        target_compile_definitions(InstalledLocalized PRIVATE
            "LVUI_EXAMPLE_STRINGS_ROOT=\"${CMAKE_CURRENT_SOURCE_DIR}\"")
    endif()
    lvui_add_application(InstalledApplication NAME DeclaredApplication
        SOURCE NativeApplication.lvui INCLUDES NativeApplicationBackend.h)
    target_include_directories(InstalledApplication PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lvui_add_application(InstalledMarkup SOURCE NativeApplicationMarkup.lvui INCLUDES NativeApplicationBackend.h)
    target_include_directories(InstalledMarkup PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lvui_add_application(InstalledForeground SOURCE NativeApplicationForeground.lvui INCLUDES NativeForegroundBackend.h)
    target_include_directories(InstalledForeground PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lvui_add_application(InstalledForegroundMarkup SOURCE NativeApplicationForegroundMarkup.lvui INCLUDES NativeForegroundBackend.h)
    target_include_directories(InstalledForegroundMarkup PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lvui_add_application(InstalledLaunch SOURCE NativeLaunchApplication.lvui INCLUDES NativeApplicationBackend.h)
    target_include_directories(InstalledLaunch PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lvui_add_application(InstalledLaunchMarkup SOURCE NativeLaunchMarkup.lvui INCLUDES NativeApplicationBackend.h)
    target_include_directories(InstalledLaunchMarkup PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lvui_add_application(InstalledLaunchScenes SOURCE NativeLaunchScenes.lvui INCLUDES NativeApplicationBackend.h)
    target_include_directories(InstalledLaunchScenes PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lvui_add_application(InstalledClose SOURCE NativeApplicationClose.lvui INCLUDES NativeApplicationBackend.h)
    target_include_directories(InstalledClose PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lvui_add_application(InstalledCloseMarkup SOURCE NativeApplicationCloseMarkup.lvui INCLUDES NativeApplicationBackend.h)
    target_include_directories(InstalledCloseMarkup PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lvui_add_application(InstalledCloseScenes SOURCE NativeApplicationCloseScenes.lvui INCLUDES NativeApplicationBackend.h)
    target_include_directories(InstalledCloseScenes PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lvui_add_application(InstalledWindowClose SOURCE NativeWindowClose.lvui
        INCLUDES NativeApplicationBackend.h WindowCloseFixtureBackend.h)
    target_include_directories(InstalledWindowClose PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lvui_add_application(InstalledSingleWindowClose SOURCE NativeSingleWindowClose.lvui
        INCLUDES NativeApplicationBackend.h WindowCloseFixtureBackend.h)
    target_include_directories(InstalledSingleWindowClose PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lvui_add_application(InstalledQuitOnly SOURCE NativeApplicationQuitOnly.lvui)
    lvui_add_application(InstalledMultiWindow SOURCE NativeMultiWindow.lvui INCLUDES NativeApplicationBackend.h)
    target_include_directories(InstalledMultiWindow PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lvui_add_application(InstalledMultiWindowMarkup SOURCE NativeMultiWindowMarkup.lvui INCLUDES NativeApplicationBackend.h)
    target_include_directories(InstalledMultiWindowMarkup PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lvui_add_application(InstalledMixedScenes SOURCE NativeMixedScenes.lvui INCLUDES NativeApplicationBackend.h)
    target_include_directories(InstalledMixedScenes PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lvui_add_application(InstalledNamedWindows SOURCE NativeNamedWindows.lvui INCLUDES NativeApplicationBackend.h)
    target_include_directories(InstalledNamedWindows PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lvui_add_application(InstalledScenes SOURCE NativeScenes.lvui INCLUDES NativeApplicationBackend.h)
    target_include_directories(InstalledScenes PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lvui_add_application(InstalledNamedWindowsMarkup SOURCE NativeNamedWindowsMarkup.lvui)
    lvui_add_application(InstalledDynamicWindows SOURCE NativeDynamicWindows.lvui INCLUDES NativeDynamicWindowsBackend.h)
    target_include_directories(InstalledDynamicWindows PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    lvui_add_application(InstalledDynamicWindowsMarkup SOURCE NativeDynamicWindowsMarkup.lvui INCLUDES NativeDynamicWindowsBackend.h)
    target_include_directories(InstalledDynamicWindowsMarkup PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
