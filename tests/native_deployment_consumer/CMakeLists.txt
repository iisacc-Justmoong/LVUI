cmake_minimum_required(VERSION 3.31)
project(LVUINativeDeployment LANGUAGES CXX)
find_package(LVUI CONFIG REQUIRED COMPONENTS Native)

if(WITH_FAKE_QT)
    add_library(LVUIFakeQt SHARED fake_qt.cpp)
    set_target_properties(LVUIFakeQt PROPERTIES OUTPUT_NAME Qt6Pretend)
    set(_extra SOURCES use_fake_qt.cpp LIBRARIES LVUIFakeQt)
elseif(WITH_COLLIDING_DEPENDENCIES)
    add_library(LVUIAliasA SHARED alias_a.cpp)
    add_library(LVUIAliasB SHARED alias_b.cpp)
    set_target_properties(LVUIAliasA PROPERTIES OUTPUT_NAME "LVUI.alias")
    set_target_properties(LVUIAliasB PROPERTIES OUTPUT_NAME "LVUI_alias")
    set(_extra SOURCES use_aliases.cpp LIBRARIES LVUIAliasA LVUIAliasB)
endif()
lvui_add_application(Deployable SOURCE "${LVUI_FIXTURE_DIR}/NativeDynamicWindows.lvui"
    INCLUDES NativeDynamicWindowsBackend.h ${_extra})
target_include_directories(Deployable PRIVATE "${LVUI_FIXTURE_DIR}")

if(BAD_INSTALL_DESTINATION)
    lvui_install_application(Deployable DESTINATION "../outside")
else()
    lvui_install_application(Deployable DESTINATION "Installed Apps - 도구" ADHOC_SIGN)
endif()
