cmake_minimum_required(VERSION 3.31)
project(LVUILocalizationConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(LVUI 0.1 CONFIG REQUIRED COMPONENTS Native Localization)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE LVUI::Localization)
enable_testing()
add_test(NAME LVUI.LocalizationConsumer COMMAND consumer)
